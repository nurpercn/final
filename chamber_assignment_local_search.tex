After MIP-CCD(\(\mathbf{s}\)), we refine \(\mathbf{x}\) by local search.
Let \((\Pi,T^{tot})=\mathrm{Schedule}(\mathbf{x},\mathbf{s})\) and let \(B_c\) be the processing time placed in chamber \(c\) by \(\Pi\). The normalized load is
\begin{equation}
\label{eq:chamber_load}
\lambda_c=\frac{B_c}{K_c}.
\end{equation}
\(\mathrm{Swap}(\mathbf{x},c,c')\) exchanges the environments of a highly loaded chamber \(c\) and a lightly loaded chamber \(c'\) with \(\lambda_c>\lambda_{c'}\).
\(\mathrm{Move}(\mathbf{x},c,c')\) assigns \(c'\) to the environment of \(c\).
A neighbor is evaluated only if it satisfies \eqref{eq:assign_one}--\eqref{eq:assign_hum}.
\(\mathrm{Schedule}\) is then called, and the neighbor is accepted if
\((T^{tot},\mathbf{1}^{\mathsf{T}}\mathbf{s})\succ_{\mathrm{lex}}(T^{tot}_{cand},\mathbf{1}^{\mathsf{T}}\mathbf{s})\).
Swap is searched first; move is used only if no swap is accepted.
The search stops when a pass accepts nothing or \(B_{\mathrm{room}}\) calls of \(\mathrm{Schedule}\) have been used.

\begin{algorithm}[H]
\caption{ChamberAssignment}
\label{alg:chamber_assignment}
\begin{algorithmic}[1]
\State \textbf{Input:} sample vector \(\mathbf{s}\)
\State \textbf{Output:} \(\mathbf{x}\)
\State \(\mathbf{x}\leftarrow\mathrm{MIP\textrm{-}CCD}(\mathbf{s})\)\label{step:ca-exact}
\State \((\Pi,T^{tot})\leftarrow\mathrm{Schedule}(\mathbf{x},\mathbf{s})\)\label{step:ca-eval}
\Repeat
    \State \(\lambda_c\leftarrow B_c/K_c\) for all \(c\in\mathcal{C}\) using \(\Pi\)
    \State \(\mathit{improved}\leftarrow\mathrm{false}\)
    \For{pairs \((c,c')\) with \(\lambda_c>\lambda_{c'}\)}
        \State \(\mathbf{x}_{cand}\leftarrow\mathrm{Swap}(\mathbf{x},c,c')\)\label{step:ca-swap}
        \If{\(\mathbf{x}_{cand}\) satisfies \eqref{eq:assign_one}--\eqref{eq:assign_hum}}
            \State \((\Pi_{cand},T^{tot}_{cand})\leftarrow\mathrm{Schedule}(\mathbf{x}_{cand},\mathbf{s})\)
            \If{\((T^{tot},\mathbf{1}^{\mathsf{T}}\mathbf{s})\succ_{\mathrm{lex}}(T^{tot}_{cand},\mathbf{1}^{\mathsf{T}}\mathbf{s})\)}
                \State \((\mathbf{x},\Pi,T^{tot})\leftarrow(\mathbf{x}_{cand},\Pi_{cand},T^{tot}_{cand})\)\label{step:ca-accept}
                \State \(\mathit{improved}\leftarrow\mathrm{true}\); \textbf{break}
            \EndIf
        \EndIf
    \EndFor
    \If{\(\mathit{improved}=\mathrm{false}\)}
        \For{pairs \((c,c')\) with \(\lambda_c>\lambda_{c'}\)}
            \State \(\mathbf{x}_{cand}\leftarrow\mathrm{Move}(\mathbf{x},c,c')\)\label{step:ca-move}
            \If{\(\mathbf{x}_{cand}\) satisfies \eqref{eq:assign_one}--\eqref{eq:assign_hum}}
                \State \((\Pi_{cand},T^{tot}_{cand})\leftarrow\mathrm{Schedule}(\mathbf{x}_{cand},\mathbf{s})\)
                \If{\((T^{tot},\mathbf{1}^{\mathsf{T}}\mathbf{s})\succ_{\mathrm{lex}}(T^{tot}_{cand},\mathbf{1}^{\mathsf{T}}\mathbf{s})\)}
                    \State \((\mathbf{x},\Pi,T^{tot})\leftarrow(\mathbf{x}_{cand},\Pi_{cand},T^{tot}_{cand})\)
                    \State \(\mathit{improved}\leftarrow\mathrm{true}\); \textbf{break}
                \EndIf
            \EndIf
        \EndFor
    \EndIf
\Until{\(\mathit{improved}=\mathrm{false}\) or \(B_{\mathrm{room}}\) calls of \(\mathrm{Schedule}\)}
\State \Return \(\mathbf{x}\)
\end{algorithmic}
\end{algorithm}

Step~\ref{step:ca-exact} solves MIP-CCD(\(\mathbf{s}\)).
Step~\ref{step:ca-eval} evaluates that assignment.
Each pass ranks chambers by \eqref{eq:chamber_load}, tries \(\mathrm{Swap}\) (Step~\ref{step:ca-swap}), and only then \(\mathrm{Move}\) (Step~\ref{step:ca-move}).
The first neighbor that satisfies the lexicographic test is accepted (Step~\ref{step:ca-accept}).
